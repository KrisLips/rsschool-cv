## Code Examples
```
function getTimeBySeconds(sec) {
  const hour = Math.trunc(sec / 3600)
    .toString()
    .padStart(2, "0");
  const minute = Math.trunc((sec - hour * 3600) / 60)
    .toString()
    .padStart(2, "0");
  const seconds = (sec - hour * 3600 - minute * 60).toString().padStart(2, "0");
  const totalsec = `${hour}:${minute}:${seconds}`;
  return totalsec;
}
```
